ESTRUCTURA ARCHIVO TÍPICO:
---------------------------

IPC Handler (src/main/ipc/algo.js):
import { ipcMain } from 'electron';
import { getDatabase } from '../database/connection.js';

export function registerAlgoHandlers() {
  ipcMain.handle('algo:create', async (event, data) => {
    const db = getDatabase();
    const stmt = db.prepare('INSERT INTO tabla (...) VALUES (...)');
    return stmt.run(data);
  });
}

Preload (src/preload/preload.js):
contextBridge.exposeInMainWorld('api', {
  algo: {
    create: (data) => ipcRenderer.invoke('algo:create', data)
  }
});

React Component (src/renderer/src/pages/Algo.jsx):
function Algo() {
  const [items, setItems] = useState([]);

  const loadItems = async () => {
    const data = await window.api.algo.getAll();
    setItems(data);
  };

  useEffect(() => { loadItems(); }, []);

  return <div>{items.map(i => <p key={i.id}>{i.nombre}</p>)}</div>;
}
